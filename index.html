// 2. 提交測驗 (寫入 A~J 欄) function submitQuiz() { userAnswers = []; wrongItems = []; let correctCount = 0; for (let i = 1; i <= currentQuestions.length; i++) { const selected = document.querySelector(`input[name="q_${i}"]:checked`); } } // 計算答對比例與分數 const accuracy = (correctCount / currentQuestions.length) * 100; const accuracyStr = accuracy.toFixed(1) + "%"; const score = Math.round(accuracy); // 格式化日期：YYYY/MM/DD const rawDate = document.getElementById('classDate').value; const formattedClassDate = rawDate.replace(/-/g, '/'); const payload = { action: 'submitQuiz', classDate: formattedClassDate, department: document.getElementById('department').value.trim(), userName: document.getElementById('userName').value.trim(), bankIds: currentQuestions.map(q => q.bankId), correctAnswers: currentQuestions.map(q => q.answer), userAnswers: userAnswers, accuracyStr: accuracyStr, score: score, feedback: document.getElementById('feedback').value.trim() }; renderReviewSection(score, accuracyStr); switchStep(2, 3); } else { showCustomModal('寫入失敗：' + resData.message); } }) .catch(err => { showLoading(false); showCustomModal('網路錯誤：' + err.message); }); } // 3. 渲染檢討與錯題簽名區域 function renderReviewSection(score, accuracyStr) { document.getElementById('scoreDisplay').innerText = `您的測驗得分：${score} 分 (答對率：${accuracyStr})`; const container = document.getElementById('reviewContainer'); container.innerHTML = ''; if (wrongItems.length === 0) { container.innerHTML = `

您有 ${wrongItems.length} 題答錯。請針對下列錯題進行檢討，並於每題下方逐條簽上您的完整工號姓名「${userName}」以示確認：

`; wrongItems.forEach((item, index) => { html += `

${item.question}

`; }); container.innerHTML = html; setTimeout(initCanvases, 100); // 綁定 Canvas 事件 } // 初始化手寫簽名板 (支援觸控與滑鼠) let canvasContexts = []; function initCanvases() { canvasContexts = []; wrongItems.forEach((_, idx) => { const cvs = document.getElementById(`canvas_${idx}`); if (!cvs) return; // 調整實際解析度 cvs.width = cvs.offsetWidth; cvs.height = cvs.offsetHeight; const ctx = cvs.getContext('2d'); ctx.lineWidth = 3; ctx.strokeStyle = '#000'; canvasContexts[idx] = { cvs, ctx, isSigned: false }; const pos = getPos(e, cvs); ctx.lineTo(pos.x, pos.y); ctx.stroke(); } function endDraw() { drawing = false; } cvs.addEventListener('mousedown', startDraw); cvs.addEventListener('mousemove', moveDraw); e.preventDefault(); }); cvs.addEventListener('touchend', endDraw); }); } function getPos(e, cvs) { const rect = cvs.getBoundingClientRect(); return { x: e.clientX - rect.left, y: e.clientY - rect.top }; } function clearCanvas(idx) { const item = canvasContexts[idx]; if (item) { item.ctx.clearRect(0, 0, item.cvs.width, item.cvs.height); item.isSigned = false; } } // 4. 提交簽名 (寫入 K~L 欄) function submitSignature() { if (wrongItems.length > 0) { for (let i = 0;
